import { useState, useRef, useEffect } from 'react'
import styles from './ActionButtons.module.scss'

function ActionButtons({
  zekrText,
  onReset,
  onClick,
  progress,
  isComplete,
  showReset = true,
  isFocusMode = false
}) {
  const [clicked, setClicked] = useState(false)
  const timerRef = useRef(null)

  const handleClick = (e) => {
    e?.preventDefault()
    e?.stopPropagation()

    // تشغيل الأنيميشن
    setClicked(true)
    clearTimeout(timerRef.current)
    timerRef.current = setTimeout(() => setClicked(false), 150)

    // تنفيذ دالة التسبيح
    onClick()
  }

  useEffect(() => {
    return () => clearTimeout(timerRef.current)
  }, [])

  return (
    <div className={`${styles.buttons} ${isFocusMode ? styles.focusButtons : ''}`}>

      {/* زر المسح: يظهر دائماً في الوضع العادي، ويختفي فقط في وضع التركيز */}
      {showReset && !isFocusMode && (
        <button
          type="button"
          className={styles.reset}
          onClick={onReset}
          disabled={!onReset} // بيتعطل لو العدد 0، بس بيفضل ظاهر
          title="تصفير"
        >
          ↺
        </button>
      )}

      {/* زر التسبيح */}
      <button
        type="button"
        className={`
          ${styles.click} 
          ${clicked ? styles.clicked : ''} 
          ${isComplete ? styles.complete : ''} 
          ${isFocusMode ? styles.floatingRightBtn : ''}
        `}
        style={{ '--progress': `${progress}%` }}
        onClick={handleClick}
        aria-label="تسبيح"
      >
        <span>{isFocusMode ? 'سبّح' : zekrText}</span>
      </button>

    </div>
  )
}

const ChevronDownIcon = () => (
  <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="6 9 12 15 18 9" />

const CheckIcon = () => (
  <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">

function AdhkarSelect({ adhkar = [], value, onChange }) {
  const [isOpen, setIsOpen] = useState(false)
  const [search, setSearch] = useState('')
  const ref = useRef(null)
  const searchInputRef = useRef(null)

  const selected = adhkar.find((z) => z.id === value)

  // فلترة الأذكار بالبحث
  const filteredAdhkar = useMemo(() => {
    if (!search.trim()) return adhkar
    return adhkar.filter((z) =>
      z.text.toLowerCase().includes(search.trim().toLowerCase())
    )
  }, [adhkar, search])

  // إغلاق القائمة عند النقر خارجها
  useEffect(() => {
    if (!isOpen) return

    const handleClickOutside = (e) => {
      if (ref.current && !ref.current.contains(e.target)) {
        setIsOpen(false)
      }
    }

    document.addEventListener('mousedown', handleClickOutside)
    return () => document.removeEventListener('mousedown', handleClickOutside)
  }, [isOpen])

  // إغلاق بـ Escape
  useEffect(() => {
    if (!isOpen) return

    const handleKeyDown = (e) => {
      if (e.code === 'Escape') setIsOpen(false)
    }

    document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [isOpen])

  // إعادة ضبط البحث وتركيز حقل البحث عند الفتح
  useEffect(() => {
    if (isOpen) {
      setSearch('')
      if (adhkar.length > 5) {
        setTimeout(() => searchInputRef.current?.focus(), 50)
      }
    }
  }, [isOpen, adhkar.length])

  const handleSelect = (id) => {
    onChange(id)
    setIsOpen(false)
  }

  return (
    <div className={styles.wrapper} ref={ref}>
      {/* زر فتح القائمة */}
      <button
        type="button"
        className={`${styles.trigger} ${isOpen ? styles.open : ''}`}
        onClick={() => setIsOpen((o) => !o)}
        aria-haspopup="listbox"
        aria-expanded={isOpen}
        aria-label="اختيار الذكر الحالي"
      >
        <span className={styles.label}>
          {selected?.text ?? 'اختر ذكرًا للبدء...'}
        </span>
        <span className={styles.arrow}>
          <ChevronDownIcon />
        </span>
      </button>

      {/* القائمة المنسدلة */}
      {isOpen && (
        <div className={styles.dropdown}>
          {/* شريط بحث يظهر لو كانت الأذكار أكثر من 5 */}
          {adhkar.length > 5 && (
            <div className={styles.searchBox}>
              <input
                ref={searchInputRef}
                type="text"
                className={styles.searchInput}
                placeholder="ابحث عن ذكر..."
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                onClick={(e) => e.stopPropagation()}
              />
            </div>
          )}

          <ul className={styles.menu} role="listbox">
            {filteredAdhkar.length === 0 ? (
              <li className={styles.empty}>لا توجد أذكار مطابقة للبحث</li>
            ) : (
              filteredAdhkar.map((zekr) => {
                const isSelected = zekr.id === value
                return (
                  <li
                    key={zekr.id}
                    role="option"
                    aria-selected={isSelected}
                    className={`${styles.item} ${isSelected ? styles.selected : ''}`}
                    onClick={() => handleSelect(zekr.id)}
                  >
                    <div className={styles.itemInfo}>
                      <span className={styles.itemText}>{zekr.text}</span>
                      {zekr.default_target ? (
                        <span className={styles.targetBadge}>
                          {zekr.default_target}
                        </span>
                      ) : null}
                    </div>

                    {isSelected && (
                      <span className={styles.checkIcon}>
                        <CheckIcon />
                      </span>
                    )}
                  </li>
                )
              })
            )}
          </ul>
        </div>
      )}
    </div>
  )
}

export default AdhkarSelect

import { useEffect, useRef } from 'react'
import { createPortal } from 'react-dom'
import styles from './FocusMode.module.scss'

const ExitFocusIcon = () => (
  <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round">
    <line x1="18" y1="6" x2="6" y2="18" />
    <line x1="6" y1="6" x2="18" y2="18" />
  </svg>
)

function FocusMode({ isOpen = true, onClose, onCount, children }) {
  const wakeLockRef = useRef(null)
  const onCloseRef = useRef(onClose)
  onCloseRef.current = onClose

  // 1. ملء الشاشة الكامل
  useEffect(() => {
    if (!isOpen) return

    const docEl = document.documentElement
    const enterFullscreen = async () => {
      try {
        if (!document.fullscreenElement && !document.webkitFullscreenElement) {
          if (docEl.requestFullscreen) {
            await docEl.requestFullscreen({ navigationUI: 'hide' })
          } else if (docEl.webkitRequestFullscreen) {
            await docEl.webkitRequestFullscreen()
          }
        }
      } catch (err) {
        console.warn('تعذر تفعيل ملء الشاشة:', err)
      }
    }

    enterFullscreen()

    const handleFullscreenChange = () => {
      const isStillFullscreen = !!(document.fullscreenElement || document.webkitFullscreenElement)
      if (!isStillFullscreen) {
        onCloseRef.current?.()
      }
    }

    document.addEventListener('fullscreenchange', handleFullscreenChange)
    document.addEventListener('webkitfullscreenchange', handleFullscreenChange)

    return () => {
      document.removeEventListener('fullscreenchange', handleFullscreenChange)
      document.removeEventListener('webkitfullscreenchange', handleFullscreenChange)

      try {
        if (document.fullscreenElement || document.webkitFullscreenElement) {
          if (document.exitFullscreen) {
            document.exitFullscreen()
          } else if (document.webkitExitFullscreen) {
            document.webkitExitFullscreen()
          }
        }
      } catch (err) { }
    }
  }, [isOpen])

  // 2. زر Escape
  useEffect(() => {
    if (!isOpen) return
    const handleKeyDown = (e) => {
      if (e.code === 'Escape') onCloseRef.current?.()
    }
    document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [isOpen])

  // 3. منع انطفاء الشاشة
  useEffect(() => {
    if (!isOpen) return
    const requestWakeLock = async () => {
      try {
        if ('wakeLock' in navigator) {
          wakeLockRef.current = await navigator.wakeLock.request('screen')
        }
      } catch (err) { }
    }
    requestWakeLock()
    return () => {
      if (wakeLockRef.current) {
        wakeLockRef.current.release().catch(() => { })
        wakeLockRef.current = null
      }
    }
  }, [isOpen])

  // 4. قفل التمرير
  useEffect(() => {
    if (!isOpen) return
    const originalOverflow = document.body.style.overflow
    document.body.style.overflow = 'hidden'
    return () => {
      document.body.style.overflow = originalOverflow
    }
  }, [isOpen])

  if (!isOpen) return null

  // ✅ التسبيح بالنقر على أي مكان في الشاشة + اهتزاز خفيف
  const handleTapScreen = (e) => {
    if (onCount) {
      onCount()
      if ('vibrate' in navigator) {
        navigator.vibrate(15) // اهتزاز خفيف جداً يؤكد التسبيحة
      }
    }
  }

  return createPortal(
    <div
      className={styles.overlay}
      onClick={handleTapScreen} // ✅ النقر في أي مكان بالشاشة يحسب تسبيحة
      role="dialog"
      aria-modal="true"
      aria-label="وضع التركيز التام"
    >
      {/* زر الخروج: نوقف انتشار الحدث stopPropagation حتى لا يحسب تسبيحة عند الخروج */}
      <button
        type="button"
        className={styles.exitBtn}
        onClick={(e) => {
          e.stopPropagation()
          onCloseRef.current?.()
        }}
        aria-label="خروج من وضع التركيز"
        title="خروج من وضع التركيز"
      >
        <ExitFocusIcon />
      </button>

      {/* محتوى الشاشة */}
      <div className={styles.centerContainer}>
        {children}
      </div>
    </div>,
    document.body
  )
}

  const [selectedGroupId, setSelectedGroupId] = useState(null)
  const [currentIndex, setCurrentIndex] = useState(0)
  const [localCount, setLocalCount] = useState(0)
  const [showTransition, setShowTransition] = useState(false)
  const [isComplete, setIsComplete] = useState(false)
  const [isFocusMode, setIsFocusMode] = useState(false) // ✅ حالة وضع التركيز للمجموعات

  // 1. اختيار أول مجموعة تلقائياً إن وُجدت
  useEffect(() => {
    if (!selectedGroupId && groups.length > 0) {
      setSelectedGroupId(groups[0].id)
    }
  }, [groups, selectedGroupId])

  const selectedGroup = groups.find((g) => g.id === selectedGroupId)
  const items = selectedGroup?.items ?? []
  const currentItem = items[currentIndex]

  // حماية الفهرس لو حُذفت عناصر
  useEffect(() => {
    if (items.length > 0 && currentIndex >= items.length) {
      setCurrentIndex(0)
      setLocalCount(0)
    }
  }, [items.length, currentIndex])

  // هدف الذكر داخل المجموعة
  const currentTarget = currentItem
    ? currentItem.target ?? currentItem.default_target ?? null
    : null

  // الخروج من وضع التركيز تلقائياً عند إتمام المجموعة كاملة
  useEffect(() => {
    if (isFocusMode && isComplete) {
      const timer = setTimeout(() => setIsFocusMode(false), 3000)
      return () => clearTimeout(timer)
    }
  }, [isFocusMode, isComplete])

  // الانتقال للذكر التالي
  const goToNext = () => {
    setShowTransition(false)
    if (currentIndex < items.length - 1) {
      setCurrentIndex((prev) => prev + 1)
      setLocalCount(0)
    } else {
      setIsComplete(true)
    }
  }

  // الرجوع للذكر السابق
  const goToPrev = () => {
    if (currentIndex > 0) {
      setShowTransition(false)
      setCurrentIndex((prev) => prev - 1)
      setLocalCount(0)
    }
  }

  // إعادة البدء
  const restartGroup = () => {
    setIsComplete(false)
    setCurrentIndex(0)
    setLocalCount(0)
  }

  // تغيير المجموعة
  const handleGroupChange = (newGroupId) => {
    setSelectedGroupId(newGroupId)
    setCurrentIndex(0)
    setLocalCount(0)
    setShowTransition(false)
    setIsComplete(false)
    setIsFocusMode(false)
  }

  // مراقبة الوصول للهدف
  useEffect(() => {
    if (!currentItem || !currentTarget) return
    if (localCount < currentTarget) return
    if (showTransition || isComplete) return

    const isLast = currentIndex === items.length - 1

    if (isLast) {
      setIsComplete(true)
      const timer = setTimeout(() => {
        restartGroup()
      }, 3000)
      return () => clearTimeout(timer)
    } else {
      setShowTransition(true)
      const timer = setTimeout(() => {
        goToNext()
      }, 2000)
      return () => clearTimeout(timer)
    }
  }, [localCount, currentTarget, currentIndex, items.length])

  const handleAddClick = () => {
    if (!currentItem) return
    if (currentTarget && localCount >= currentTarget) return
    setLocalCount((prev) => prev + 1)
    onAddClick(currentItem.zekr_id)
  }

  const handleReset = () => {
    if (!currentItem) return
    setLocalCount(0)
    onResetClick(currentItem.zekr_id)
  }

  const handleSetTarget = () => {
    if (!currentItem || !selectedGroup) return
    onSetTarget(selectedGroup.id, currentItem.id, currentTarget)
  }

  if (groups.length === 0) {
    return (
      <div className={styles.emptyView}>
        <div className={styles.emptyIcon}>📂</div>
        <h3>لا توجد مجموعات أذكار بعد</h3>
        <p>قم بإنشاء مجموعتك الأولى من تبويب الإعدادات لتبدأ التسبيح التلقائي.</p>
      </div>
    )
  }

  // محتوى المجموعة التفاعلي (مشترك بين العرض العادي ووضع التركيز)
  const renderGroupContent = () => {
    if (items.length === 0) {
      return (
        <div className={styles.emptyGroupCard}>
          <p>هذه المجموعة لا تحتوي على أذكار حتى الآن.</p>
          <span>أضف أذكاراً إليها من صفحة الإعدادات.</span>
        </div>
      )
    }

    if (isComplete) {
      return (
        <div className={styles.completeCard} onClick={restartGroup} title="اضغط للبدء من جديد">
          <div className={styles.celebrateIcon}>🎉</div>
          <h2 className={styles.completeTitle}>تقبل الله! أتممت الورد</h2>
          <p className={styles.groupName}>{selectedGroup?.name}</p>
          <button type="button" className={styles.restartBtn} onClick={restartGroup}>
            إعادة المجموعة الآن ↺
          </button>
        </div>
      )
    }

    if (showTransition) {
      return (
        <div className={styles.transitionCard} onClick={goToNext} title="اضغط للمتابعة فوراً">
          <div className={styles.doneIcon}>✓</div>
          <h2 className={styles.transitionTitle}>أحسنت! أتممت الذكر</h2>
          <p className={styles.nextPreview}>
            التالي: <strong>{items[currentIndex + 1]?.text ?? ''}</strong>
          </p>
          <span className={styles.skipPrompt}>اضغط في أي مكان للمتابعة فوراً ⏭</span>
        </div>
      )
    }

    return (
      <div className={styles.content}>
        {/* شريط التقدم المجزأ للورد */}
        <div className={styles.storyProgress} aria-label="تقدم المجموعة">
          {items.map((_, idx) => (
            <div
              key={idx}
              className={`${styles.progressBar} ${idx < currentIndex
                ? styles.barCompleted
                : idx === currentIndex
                  ? styles.barActive
                  : ''
                }`}
            />
          ))}
        </div>

        {/* أزرار التنقل تظهر خارج وضع التركيز */}
        {!isFocusMode && (
          <div className={styles.navBar}>
            <button
              type="button"
              className={styles.navBtn}
              onClick={goToPrev}
              disabled={currentIndex === 0}
              title="الذكر السابق"
            >
              <ArrowRightIcon />
              <span>السابق</span>
            </button>

            <span className={styles.stepIndicator}>
              الذكر {currentIndex + 1} من {items.length}
            </span>

            <button
              type="button"
              className={styles.navBtn}
              onClick={goToNext}
              disabled={currentIndex === items.length - 1}
              title="تخطي للذكر التالي"
            >
              <span>تخطي</span>
              <ArrowLeftIcon />
            </button>
          </div>
        )}

        {/* عرض الذكر والعداد */}
        <ZekrDisplay
          zekrText={currentItem.text}
          count={localCount}
          target={currentTarget}
          onClickCount={handleSetTarget}
          isComplete={!!currentTarget && localCount >= currentTarget}
          isFocusMode={isFocusMode}
        />

        {!currentTarget && (
          <div className={styles.openTargetAlert}>
            <span>هذا الذكر بدون هدف محدد. اضغط "تخطي" عند الانتهاء منه.</span>
          </div>
        )}

        {/* أزرار التسبيح */}
        <ActionButtons
          zekrText={currentItem.text}
          onReset={handleReset}
          onClick={handleAddClick}
          progress={
            currentTarget
              ? Math.min((localCount / currentTarget) * 100, 100)
              : 0
          }
          isComplete={!!currentTarget && localCount >= currentTarget}
          isFocusMode={isFocusMode}
          showReset={!isFocusMode}
        />

        {/* زر تفعيل وضع التركيز للمجموعة */}
        {!isFocusMode && items.length > 0 && (
          <button
            type="button"
            className={styles.focusBtn}
            onClick={() => setIsFocusMode(true)}
            title="الانتقال لوضع التركيز بدون مشتتات"
          >
            🌙 وضع التركيز
          </button>
        )}
      </div>
    )
  }

  return (
    <div className={styles.view}>
      {/* قائمة اختيار المجموعة تختفي أثناء وضع التركيز لمنع التشتيت */}
      {!isFocusMode && (
        <div className={styles.selectWrapper}>
          <GroupSelect
            groups={groups}
            value={selectedGroupId}
            onChange={handleGroupChange}
          />
        </div>
      )}

      {/* التبديل بين وضع التركيز والعرض العادي */}
      {isFocusMode ? (
        <FocusMode
          onCount={handleAddClick}
          onClose={() => setIsFocusMode(false)}>
          {renderGroupContent()}
        </FocusMode>
      ) : (
        renderGroupContent()
      )}
    </div>
  )
}

function GroupSelect({ groups = [], value, onChange }) {
  const [isOpen, setIsOpen] = useState(false)
  const ref = useRef(null)

  // العثور على المجموعة المحددة حالياً
  const selected = groups.find((g) => g.id === value)

  // إغلاق القائمة عند النقر خارجها
  useEffect(() => {
    if (!isOpen) return

    const handleClickOutside = (e) => {
      if (ref.current && !ref.current.contains(e.target)) {
        setIsOpen(false)
      }
    }

    document.addEventListener('mousedown', handleClickOutside)
    return () => document.removeEventListener('mousedown', handleClickOutside)
  }, [isOpen])

  // إغلاق بـ Escape
  useEffect(() => {
    if (!isOpen) return

    const handleKeyDown = (e) => {
      if (e.code === 'Escape') setIsOpen(false)
    }

    document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [isOpen])

  if (groups.length === 0) {
    return (
      <div className={styles.empty}>
        📂 لا توجد مجموعات أذكار. أضِف واحدة من صفحة الإعدادات.
      </div>
    )
  }

  const handleSelect = (group) => {
    onChange(group.id) // نرسل الـ ID بنوعه الأصلي (رقم أو نص) دون تشويه
    setIsOpen(false)
  }

  return (
    <div className={styles.wrapper} ref={ref}>
      {/* زر فتح القائمة */}
      <button
        type="button"
        className={`${styles.trigger} ${isOpen ? styles.open : ''}`}
        onClick={() => setIsOpen((o) => !o)}
        aria-haspopup="listbox"
        aria-expanded={isOpen}
        aria-label="اختيار مجموعة الأذكار"
      >
        <div className={styles.triggerContent}>
          <span className={styles.folderIcon}>
            <FolderIcon />
          </span>
          <span className={styles.label}>
            {selected ? selected.name : 'اختر مجموعة أذكار للبدء...'}
          </span>
          {selected && (
            <span className={styles.countBadge}>
              {selected.items?.length ?? 0} ذكر
            </span>
          )}
        </div>

        <span className={styles.arrow}>
          <ChevronDownIcon />
        </span>
      </button>

      {/* القائمة المنسدلة */}
      {isOpen && (
        <div className={styles.dropdown}>
          <ul className={styles.menu} role="listbox">
            {groups.map((g) => {
              const isSelected = g.id === value
              const itemCount = g.items?.length ?? 0
              return (
                <li
                  key={g.id}
                  role="option"
                  aria-selected={isSelected}
                  className={`${styles.item} ${isSelected ? styles.selected : ''}`}
                  onClick={() => handleSelect(g)}
                >
                  <div className={styles.itemInfo}>
                    <div className={styles.titleLine}>
                      <span className={styles.name}>{g.name}</span>
                      <span className={styles.itemCountBadge}>
                        {itemCount} ذكر
                      </span>
                    </div>

                    {g.description && (
                      <span className={styles.desc}>{g.description}</span>
                    )}
                  </div>

                  {isSelected && (
                    <span className={styles.checkIcon}>
                      <CheckIcon />
                    </span>
                  )}
                </li>
              )
            })}
          </ul>
        </div>
      )}
    </div>
  )
}

const CheckIcon = () => (
  <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="20 6 9 17 4 12" />
  </svg>
)

function ZekrDisplay({ zekrText, count = 0, target, onClickCount, isComplete, isFocusMode = false }) {
  const [copied, setCopied] = useState(false)

  // نسخ نص الذكر للحافظة
  const handleCopy = async (e) => {
    e.stopPropagation()
    if (!zekrText) return
    try {
      await navigator.clipboard.writeText(zekrText)
      setCopied(true)
      setTimeout(() => setCopied(false), 1800)
    } catch (err) {
      console.error('فشل النسخ:', err)
    }
  }

  return (
    <div className={`
        ${styles.displayCard} 
        ${isComplete ? styles.completeGlow : ''} 
        ${isFocusMode ? styles.focusDisplay : ''} 
      `}>
      {/* بطاقة نص الذكر مع زر النسخ */}
      <div className={styles.textContainer}>
        <h2 className={styles.text}>{zekrText}</h2>

        <button
          type="button"
          className={`${styles.copyBtn} ${copied ? styles.copied : ''}`}
          onClick={handleCopy}
          title="نسخ نص الذكر"
          aria-label="نسخ نص الذكر"
        >
          {copied ? (
            <>
              <CheckIcon />
              <span className={styles.copyText}>تم النسخ</span>
            </>
          ) : (
            <>
              <CopyIcon />
              <span className={styles.copyText}>نسخ</span>
            </>
          )}
        </button>
      </div>

      {/* قسم العداد والهدف التفاعلي */}
      <div className={styles.counterSection}>
        {/* الرقم الحالي (ضخم وواضح) */}
        <div className={`${styles.currentNumber} ${isComplete ? styles.completeNumber : ''}`}>
          {count.toLocaleString('en')}
        </div>

        {/* شارة الهدف القابلة للنقر للتعديل */}
        {target ? (
          <button
            type="button"
            className={`${styles.targetBadge} ${isComplete ? styles.targetComplete : ''}`}
            onClick={onClickCount}
            title="اضغط لتعديل الهدف"
            aria-label={`الهدف الحالي ${target}، اضغط للتعديل`}
          >
            <span>🎯 الهدف: {target.toLocaleString('en')}</span>
            <span className={styles.editIcon}>
              <EditIcon />
            </span>
          </button>
        ) : (
          <button
            type="button"
            className={styles.setTargetBtn}
            onClick={onClickCount}
            title="اضغط لتحديد هدف لهذا الذكر"
          >
            <span>+ تحديد هدف</span>
          </button>
        )}
      </div>
    </div>
  )
}
