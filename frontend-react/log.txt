function AboutSection() {

      text: 'تطبيق ذِكر - سبحة إلكترونية لمتابعة وإحصاء أذكارك اليومية 📿',

        await navigator.share(shareData)
      } catch (err) {
        // تم إلغاء المشاركة بواسطة المستخدم

  return (
    <SettingsSection title="عن التطبيق">
      <div className={styles.aboutCard}>
        {/* رأس البطاقة: الشعار والمعلومات الأساسية */}
        <div className={styles.header}>
          <div className={styles.logoBadge}>📿</div>
          <div className={styles.appMeta}>
            <div className={styles.titleRow}>
              <h3 className={styles.appName}>ذِكر</h3>
              <span className={styles.versionBadge}>v1.0.0</span>
            </div>
            <p className={styles.appDesc}>
              سبحة إلكترونية ذكية تتيح لك التسبيح بتركيز ومتابعة إحصائيات أذكارك يومياً.
            </p>
          </div>
        </div>

        {/* مميزات التطبيق (Features Chips) */}
        <div className={styles.features}>
          <div className={styles.featureItem}>
            <span className={styles.featureIcon}>✨</span>
            <span>بدون إعلانات</span>
          </div>
          <div className={styles.featureItem}>
            <span className={styles.featureIcon}>🔒</span>
            <span>بياناتك محفوظة بأمان</span>
          </div>
          <div className={styles.featureItem}>
            <span className={styles.featureIcon}>⚡</span>
            <span>يعمل بدون إنترنت</span>
          </div>
        </div>

        {/* زر المشاركة */}

          <button
            type="button"

            onClick={handleShare}

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './AdhkarFormModal.module.scss'

// الأهداف الأكثر شيوعاً في الأذكار
const PRESET_TARGETS = [
  { label: 'مفتوح (بدون)', value: '' },
  { label: '33', value: '33' },
  { label: '100', value: '100' },
  { label: '1000', value: '1000' },
]

function AdhkarFormModal({ isOpen, onClose, onConfirm, initial = null }) {
  const [text, setText] = useState('')
  const [target, setTarget] = useState('')
  const [error, setError] = useState('')
  const [isSubmitting, setIsSubmitting] = useState(false)

  useEffect(() => {
    if (isOpen) {
      setText(initial?.text ?? '')
      setTarget(initial?.default_target ? String(initial.default_target) : '')
      setError('')
      setIsSubmitting(false)
    }
  }, [isOpen, initial])

  const handleSubmit = async (e) => {
    e.preventDefault()
    const trimmedText = text.trim()

    if (!trimmedText) {
      setError('الرجاء إدخال نص الذكر')
      return
    }

    let targetValue = null
    if (target !== '') {
      const num = Number(target)
      // التأكد من أنه عدد صحيح ضمن النطاق
      if (!Number.isInteger(num) || num < 1 || num > 10000) {
        setError('الهدف يجب أن يكون عدداً صحيحاً بين 1 و 10,000')
        return
      }
      targetValue = num
    }

    try {
      setIsSubmitting(true)
      await onConfirm(trimmedText, targetValue)
    } catch (err) {
      setError(err?.message || 'حدث خطأ أثناء الحفظ، يرجى المحاولة ثانية')
    } finally {
      setIsSubmitting(false)
    }
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={initial ? 'تعديل الذكر' : 'إضافة ذكر جديد'}
    >
      <form className={styles.form} onSubmit={handleSubmit} noValidate>
        {/* حقل نص الذكر */}
        <div className={styles.fieldGroup}>
          <label htmlFor="zekr-text" className={styles.label}>
            <span>نص الذكر أو الدعاء</span>
            <span className={styles.required}>*</span>
          </label>
          <textarea
            id="zekr-text"
            className={styles.textarea}
            value={text}
            onChange={(e) => {
              setText(e.target.value)
              if (error) setError('')
            }}
            placeholder="اكتب الذكر هنا (مثال: أستغفر الله العظيم وأتوب إليه)..."
            rows={3}
            autoFocus
          />
        </div>

        {/* حقل الهدف */}
        <div className={styles.fieldGroup}>
          <div className={styles.labelRow}>
            <label htmlFor="zekr-target" className={styles.label}>
              الهدف المطلوب
            </label>
            <span className={styles.optionalHint}>اختياري</span>
          </div>

          {/* أزرار سريعة لاختيار الهدف الشائع بنقرة واحدة */}
          <div className={styles.presetChips}>
            {PRESET_TARGETS.map((preset) => {
              const isSelected = String(target) === String(preset.value)
              return (
                <button
                  key={preset.label}
                  type="button"
                  className={`${styles.presetBtn} ${isSelected ? styles.activePreset : ''}`}
                  onClick={() => {
                    setTarget(preset.value)
                    if (error) setError('')
                  }}
                >
                  {preset.label}
                </button>
              )
            })}
          </div>

          <input
            id="zekr-target"
            className={styles.input}
            type="number"
            value={target}
            onChange={(e) => {
              setTarget(e.target.value)
              if (error) setError('')
            }}
            placeholder="أو اكتب رقمًا مخصصًا (مثال: 70)"
            min={1}
            max={10000}
          />
        </div>

        {/* رسالة الخطأ */}
        {error && (
          <div className={styles.error} role="alert">
            <span className={styles.errorIcon}>⚠️</span>
            <span>{error}</span>
          </div>
        )}

        {/* أزرار الإجراءات */}
        <div className={styles.buttons}>
          <button
            type="button"
            className={styles.cancel}
            onClick={onClose}
            disabled={isSubmitting}
          >
            إلغاء
          </button>
          <button
            type="submit"
            className={styles.confirm}
            disabled={isSubmitting}
          >
            {isSubmitting ? 'جاري الحفظ...' : initial ? 'حفظ التعديلات' : 'إضافة'}
          </button>
        </div>
      </form>
    </Modal>
  )
}

import { useState, useMemo } from 'react'

const EditIcon = () => (
  <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
    <path d="M11 4H4a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-7" />

const DeleteIcon = () => (
  <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="3 6 5 6 21 6" />

function AdhkarSection({ adhkar = [], onAdd, onUpdate, onDelete }) {
  const [formMode, setFormMode] = useState(null) // null | 'add' | { editing: zekr }
  const [deleteTarget, setDeleteTarget] = useState(null)
  const [isDeleting, setIsDeleting] = useState(false)
  const [search, setSearch] = useState('')

  // فلترة الأذكار بالبحث
  const filteredAdhkar = useMemo(() => {
    if (!search.trim()) return adhkar
    return adhkar.filter((z) =>
      z.text.toLowerCase().includes(search.trim().toLowerCase())
    )
  }, [adhkar, search])

  const handleAdd = async (text, target) => {
    await onAdd(text, target)
    setFormMode(null)
  }

  const handleUpdate = async (text, target) => {
    await onUpdate(formMode.editing.id, text, target)
    setFormMode(null)
  }

  const handleDelete = async () => {
    if (!deleteTarget) return
    try {
      setIsDeleting(true)
      await onDelete(deleteTarget.id)
      setDeleteTarget(null)
    } catch (err) {
      console.error('فشل حذف الذكر:', err)
    } finally {
      setIsDeleting(false)
    }
  }

  return (
    <SettingsSection title={`الأذكار (${adhkar.length})`}>
      <div className={styles.sectionContainer}>
        {/* شريط الإجراءات العلوي: زر الإضافة وشريط البحث */}
        <div className={styles.topBar}>
          <button
            type="button"
            className={styles.addBtn}
            onClick={() => setFormMode('add')}
          >
            <span>+</span>
            <span>إضافة ذكر جديد</span>
          </button>

          {adhkar.length > 4 && (
            <input
              type="text"
              className={styles.searchInput}
              placeholder="بحث في الأذكار..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
            />
          )}
        </div>

        {/* قائمة الأذكار */}
        {filteredAdhkar.length > 0 ? (
          <ul className={styles.list}>
            {filteredAdhkar.map((z) => (
              <li key={z.id} className={styles.item}>
                <div className={styles.info}>
                  <span className={styles.text}>{z.text}</span>
                  {z.default_target ? (
                    <span className={styles.targetBadge}>
                      🎯 الهدف: {z.default_target.toLocaleString('en')}
                    </span>
                  ) : (
                    <span className={styles.noTargetBadge}>مفتوح</span>
                  )}
                </div>

                <div className={styles.actions}>
                  <button
                    type="button"
                    className={`${styles.actionBtn} ${styles.edit}`}
                    onClick={() => setFormMode({ editing: z })}
                    title="تعديل"
                    aria-label={`تعديل ${z.text}`}
                  >
                    <EditIcon />
                  </button>
                  <button
                    type="button"
                    className={`${styles.actionBtn} ${styles.danger}`}
                    onClick={() => setDeleteTarget(z)}
                    title="حذف"
                    aria-label={`حذف ${z.text}`}
                  >
                    <DeleteIcon />
                  </button>
                </div>
              </li>
            ))}
          </ul>
        ) : (
          <div className={styles.emptyState}>
            <div className={styles.emptyIcon}>📿</div>
            <p className={styles.emptyText}>
              {search ? 'لا توجد أذكار تطابق بحثك' : 'لم تقم بإضافة أي أذكار بعد'}
            </p>
          </div>
        )}
      </div>

      {/* نوافذ الإدخال والتأكيد */}
      <AdhkarFormModal
        isOpen={formMode === 'add' || formMode?.editing != null}
        onClose={() => setFormMode(null)}
        onConfirm={formMode === 'add' ? handleAdd : handleUpdate}
        initial={formMode?.editing ?? null}
      />

      <ConfirmModal
        isOpen={deleteTarget !== null}
        onClose={() => setDeleteTarget(null)}
        onConfirm={handleDelete}
        title="حذف الذكر؟"
        message={`سيتم حذف "${deleteTarget?.text}" نهائياً مع كافة الأعداد والإحصائيات الخاصة به.`}
        confirmLabel={isDeleting ? 'جاري الحذف...' : 'نعم، احذف'}
        cancelLabel="إلغاء"
        danger
      />
    </SettingsSection>
  )
}

function AppearanceSection() {
  const { theme, toggleTheme, setTheme } = useTheme()

  const handleSelectTheme = (selectedTheme) => {
    if (theme === selectedTheme) return
    if (setTheme) {
      setTheme(selectedTheme)
    } else {
      toggleTheme()
    }
  }

  return (
    <SettingsSection title="المظهر">
      <div className={styles.container}>
        <p className={styles.hint}>
          اختر المظهر المريح لعينيك أثناء التسبيح والقراءة:
        </p>

        {/* شبكة خيارات الثيم المرئية */}
        <div className={styles.grid} role="radiogroup" aria-label="اختيار المظهر">

          {/* خيار الوضع الفاتح */}
          <button
            type="button"
            role="radio"
            aria-checked={theme === 'light'}
            className={`${styles.card} ${theme === 'light' ? styles.active : ''} ${styles.lightCard}`}
            onClick={() => handleSelectTheme('light')}
          >
            {/* معاينة بصرية مصغرة لواجهة التطبيق */}
            <div className={styles.preview}>
              <div className={styles.previewHeader} />
              <div className={styles.previewCircle} />
              <div className={styles.previewBar} />
            </div>

            <div className={styles.cardFooter}>
              <div className={styles.cardLabel}>
                <span className={styles.icon}><SunIcon /></span>
                <span>فاتح</span>
              </div>
              {theme === 'light' && (
                <div className={styles.checkBadge}>
                  <CheckIcon />
                </div>
              )}
            </div>
          </button>

          {/* خيار الوضع الداكن */}
          <button
            type="button"
            role="radio"
            aria-checked={theme === 'dark'}
            className={`${styles.card} ${theme === 'dark' ? styles.active : ''} ${styles.darkCard}`}
            onClick={() => handleSelectTheme('dark')}
          >
            {/* معاينة بصرية مصغرة لواجهة التطبيق الداكنة */}
            <div className={styles.preview}>

function DataSection({
  onDeleteCounts,
  onDeleteTargets,
  onDeleteClicks,
  onDeleteAll,
  onExportBackup,
  onImportBackup,
}) {
  const [confirmType, setConfirmType] = useState(null)
  const [loading, setLoading] = useState(false)
  const [importStatus, setImportStatus] = useState(null) // رسائل نجاح أو خطأ
  const [pendingFileContent, setPendingFileContent] = useState(null)

  const fileInputRef = useRef(null)

  const actionHandlers = {
    counts: onDeleteCounts,
    targets: onDeleteTargets,
    clicks: onDeleteClicks,
    all: onDeleteAll,
  }

  const currentConfirmData = confirmType === 'all'
    ? DANGER_OPTION
    : RESET_OPTIONS.find((opt) => opt.id === confirmType)

  const handleConfirm = async () => {
    if (!confirmType || !actionHandlers[confirmType]) return

    setLoading(true)
    try {
      await actionHandlers[confirmType]()
      setConfirmType(null)
    } catch (err) {
      console.error('فشل تنفيذ عملية المسح:', err)
    } finally {
      setLoading(false)
    }
  }

  // عند اختيار المستخدم لملف النسخة الاحتياطية
  const handleFileChange = (e) => {
    const file = e.target.files?.[0]
    if (!file) return

    const reader = new FileReader()
    reader.onload = (event) => {
      setPendingFileContent(event.target.result)
    }
    reader.readAsText(file)

    // تفريغ الحقل لاختياره مجدداً مستقبلاً
    e.target.value = ''
  }

  // تنفيذ استيراد النسخة بعد التأكيد
  const handleConfirmImport = async () => {
    if (!pendingFileContent) return

    setLoading(true)
    setImportStatus(null)
    try {
      const result = await onImportBackup(pendingFileContent)
      setImportStatus({
        type: 'success',
        text: `تم استرجاع البيانات بنجاح (${result.count} ذكر ومجموعاتها).`,
      })
      setPendingFileContent(null)
    } catch (err) {
      setImportStatus({
        type: 'error',
        text: err.message || 'فشل استيراد النسخة الاحتياطية، تأكد من صحة الملف.',
      })
      setPendingFileContent(null)
    } finally {
      setLoading(false)
    }
  }

  return (
    <SettingsSection title="إدارة البيانات والذاكرة">
      <div className={styles.container}>

        {/* ===================================
            1. قسم النسخ الاحتياطي (Backup & Restore)
        =================================== */}
        <div className={styles.backupSection}>
          <h4 className={styles.subSectionTitle}>النسخ الاحتياطي السحابي والمحلي</h4>

          <div className={styles.backupGrid}>
            {/* بطاقة التصدير */}
            <div className={styles.backupCard}>
              <div className={styles.backupHeader}>
                <span className={styles.backupIcon}>📥</span>
                <div>
                  <span className={styles.backupTitle}>تصدير نسخة احتياطية</span>
                  <p className={styles.backupDesc}>
                    حفظ ملف JSON يحوي كل أذكارك، أهدافك، ومجموعاتك لنقلها لأي جهاز.
                  </p>
                </div>
              </div>
              <button
                type="button"
                className={styles.exportBtn}
                onClick={onExportBackup}
              >
                تنزيل الملف
              </button>
            </div>

            {/* بطاقة الاستيراد */}
            <div className={styles.backupCard}>
              <div className={styles.backupHeader}>
                <span className={styles.backupIcon}>📤</span>
                <div>
                  <span className={styles.backupTitle}>استيراد نسخة سابقة</span>
                  <p className={styles.backupDesc}>
                    استرجاع الأذكار والمجموعات من ملف JSON محفوظ مسبقاً.
                  </p>
                </div>
              </div>

              {/* حقل رفع الملف غير المرئي */}
              <input
                ref={fileInputRef}
                type="file"
                accept=".json"
                style={{ display: 'none' }}
                onChange={handleFileChange}
              />

              <button
                type="button"
                className={styles.importBtn}
                onClick={() => fileInputRef.current?.click()}
              >
                اختيار ملف
              </button>
            </div>
          </div>

          {/* رسالة حالة الاستيراد */}
          {importStatus && (
            <div
              className={`${styles.statusMessage} ${importStatus.type === 'success' ? styles.statusSuccess : styles.statusError
                }`}
            >
              {importStatus.type === 'success' ? '✓ ' : '⚠️ '}
              {importStatus.text}
            </div>
          )}
        </div>

        {/* ===================================
            2. العمليات الجزئية والتصفير
        =================================== */}
        <div className={styles.optionsList}>
          <h4 className={styles.subSectionTitle}>خيارات التصفير السريع</h4>
          {RESET_OPTIONS.map((item) => (
            <div key={item.id} className={styles.row}>
              <div className={styles.rowInfo}>
                <div className={styles.titleLine}>
                  <span className={styles.icon}>{item.icon}</span>
                  <span className={styles.title}>{item.title}</span>
                </div>
                <p className={styles.desc}>{item.desc}</p>
              </div>

              <button
                type="button"
                className={styles.actionBtn}
                onClick={() => setConfirmType(item.id)}
              >
                {item.btnLabel}
              </button>
            </div>
          ))}
        </div>

        {/* ===================================
            3. منطقة الخطر (إعادة ضبط شاملة)
        =================================== */}
        <div className={styles.dangerZone}>
          <div className={styles.dangerHeader}>
            <span className={styles.dangerIcon}>{DANGER_OPTION.icon}</span>
            <div className={styles.dangerInfo}>
              <span className={styles.dangerTitle}>{DANGER_OPTION.title}</span>
              <p className={styles.dangerDesc}>{DANGER_OPTION.desc}</p>
            </div>
          </div>

          <button
            type="button"
            className={styles.dangerBtn}
            onClick={() => setConfirmType(DANGER_OPTION.id)}
          >
            {DANGER_OPTION.btnLabel}
          </button>
        </div>

      </div>

      {/* نافذة تأكيد عمليات المسح */}
      <ConfirmModal
        isOpen={confirmType !== null}
        onClose={() => setConfirmType(null)}
        onConfirm={handleConfirm}
        title={currentConfirmData?.confirmTitle ?? ''}
        message={currentConfirmData?.confirmMessage ?? ''}
        confirmLabel={loading ? 'جاري المسح...' : 'نعم، امسح'}
        cancelLabel="إلغاء"
        danger
      />

      {/* نافذة تأكيد استيراد ملف النسخة الاحتياطية */}
      <ConfirmModal
        isOpen={pendingFileContent !== null}
        onClose={() => setPendingFileContent(null)}
        onConfirm={handleConfirmImport}
        title="استيراد النسخة الاحتياطية؟"
        message="سيتم إضافة الأذكار والمجموعات الموجودة بالملف إلى تطبيقك. هل تريد المتابعة؟"
        confirmLabel={loading ? 'جاري الاستيراد...' : 'نعم، استورد الآن'}
        cancelLabel="إلغاء"
      />
    </SettingsSection>
  )
}

function GroupEditorModal({
  isOpen,
  onClose,
  group,
  allAdhkar = [],
  onAddItem,
  onUpdateItem,
  onDeleteItem,
}) {
  const [editingTargetId, setEditingTargetId] = useState(null)
  const [tempTarget, setTempTarget] = useState('')
  const [targetError, setTargetError] = useState('')
  const [searchAvailable, setSearchAvailable] = useState('')
  const [busyId, setBusyId] = useState(null)

  useEffect(() => {
    if (!isOpen) {
      setEditingTargetId(null)
      setTempTarget('')
      setTargetError('')
      setSearchAvailable('')
      setBusyId(null)
    }
  }, [isOpen])

  // حماية آمنة لعناصر المجموعة
  const items = group?.items ?? []

  // استخراج الأذكار المتاحة وفلترتها بالبحث
  const usedIds = useMemo(() => new Set(items.map((i) => i.zekr_id)), [items])

  const filteredAvailableAdhkar = useMemo(() => {
    const available = allAdhkar.filter((z) => !usedIds.has(z.id))
    if (!searchAvailable.trim()) return available
    return available.filter((z) =>
      z.text.toLowerCase().includes(searchAvailable.trim().toLowerCase())
    )
  }, [allAdhkar, usedIds, searchAvailable])

  if (!group) return null

  // ===========================
  // Handlers
  // ===========================

  const handleAdd = async (zekr_id) => {
    setBusyId(`add-${zekr_id}`)
    try {
      const zekr = allAdhkar.find((z) => z.id === zekr_id)
      const target = zekr?.default_target ?? null
      await onAddItem(group.id, zekr_id, target)
    } finally {
      setBusyId(null)
    }
  }

  const handleDelete = async (itemId) => {
    setBusyId(`del-${itemId}`)
    try {
      await onDeleteItem(group.id, itemId)
    } finally {
      setBusyId(null)
    }
  }

  const handleMoveUp = async (item, index) => {
    if (index === 0 || busyId) return
    const prev = items[index - 1]
    setBusyId(`move-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, { position: prev.position ?? index - 1 })
      await onUpdateItem(group.id, prev.id, { position: item.position ?? index })
    } finally {
      setBusyId(null)
    }
  }

  const handleMoveDown = async (item, index) => {
    if (index === items.length - 1 || busyId) return
    const next = items[index + 1]
    setBusyId(`move-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, { position: next.position ?? index + 1 })
      await onUpdateItem(group.id, next.id, { position: item.position ?? index })
    } finally {
      setBusyId(null)
    }
  }

  const handleEditTarget = (item) => {
    setEditingTargetId(item.id)
    setTempTarget(item.target != null ? String(item.target) : '')
    setTargetError('')
  }

  const handleSaveTarget = async (item) => {
    let finalTarget = null

    if (tempTarget.trim() !== '') {
      const num = Number(tempTarget)
      if (!Number.isInteger(num) || num < 1 || num > 10000) {
        setTargetError('بين 1 و 10,000')
        return
      }
      finalTarget = num
    }

    setBusyId(`target-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, { target: finalTarget })
      setEditingTargetId(null)
      setTempTarget('')
      setTargetError('')
    } finally {
      setBusyId(null)
    }
  }

  const handleCancelTarget = () => {
    setEditingTargetId(null)
    setTempTarget('')
    setTargetError('')
  }

  return (
    <Modal isOpen={isOpen} onClose={onClose} title={group.name}>
      <div className={styles.body}>
        {group.description && (
          <p className={styles.description}>{group.description}</p>
        )}

        {/* قسم أذكار المجموعة الحالية */}
        <div className={styles.section}>
          <div className={styles.sectionHeader}>
            <h4 className={styles.sectionTitle}>
              أذكار المجموعة <span className={styles.badge}>{items.length}</span>
            </h4>
          </div>

          {items.length === 0 ? (
            <div className={styles.emptyCard}>
              لا توجد أذكار مضافة لهذه المجموعة حتى الآن.
            </div>
          ) : (
            <ul className={styles.itemsList}>
              {items.map((item, index) => {
                const isCurrentBusy = busyId === `move-${item.id}` || busyId === `del-${item.id}`
                return (
                  <li
                    key={item.id}
                    className={`${styles.item} ${isCurrentBusy ? styles.isBusy : ''}`}
                  >
                    {/* رقم الترتيب */}
                    <div className={styles.positionBadge}>{index + 1}</div>

                    <div className={styles.itemInfo}>
                      <span className={styles.itemText}>{item.text}</span>

                      {/* تعديل الهدف أو عرضه */}
                      {editingTargetId === item.id ? (
                        <div className={styles.targetEditGroup}>
                          <input
                            className={`${styles.targetInput} ${targetError ? styles.hasError : ''}`}
                            type="number"
                            value={tempTarget}
                            onChange={(e) => {
                              setTempTarget(e.target.value)
                              if (targetError) setTargetError('')
                            }}
                            onKeyDown={(e) => {
                              if (e.key === 'Enter') handleSaveTarget(item)
                              if (e.key === 'Escape') handleCancelTarget()
                            }}
                            autoFocus
                            placeholder="افتراضي"
                            min={1}
                            max={10000}
                          />
                          <button
                            type="button"
                            className={styles.saveBtn}
                            onClick={() => handleSaveTarget(item)}
                            title="حفظ"
                          >
                            <CheckIcon />
                          </button>
                          <button
                            type="button"
                            className={styles.cancelBtn}
                            onClick={handleCancelTarget}
                            title="إلغاء"
                          >
                            <CloseIcon />
                          </button>
                          {targetError && (
                            <span className={styles.inlineError}>{targetError}</span>
                          )}
                        </div>
                      ) : (
                        <button
                          type="button"
                          className={styles.targetDisplay}
                          onClick={() => handleEditTarget(item)}
                          title="اضغط لتعديل هدف هذا الذكر"
                        >
                          🎯 الهدف: {item.target ?? item.default_target ?? 'مفتوح'}
                        </button>
                      )}
                    </div>

                    {/* أزرار الترتيب والحذف */}
                    <div className={styles.itemActions}>
                      <button
                        type="button"
                        className={styles.iconBtn}
                        onClick={() => handleMoveUp(item, index)}
                        disabled={index === 0 || !!busyId}
                        title="تحريك لأعلى"
                      >
                        <ArrowUpIcon />
                      </button>
                      <button
                        type="button"
                        className={styles.iconBtn}
                        onClick={() => handleMoveDown(item, index)}
                        disabled={index === items.length - 1 || !!busyId}
                        title="تحريك لأسفل"
                      >
                        <ArrowDownIcon />
                      </button>
                      <button
                        type="button"
                        className={`${styles.iconBtn} ${styles.danger}`}
                        onClick={() => handleDelete(item.id)}
                        disabled={!!busyId}
                        title="إزالة من المجموعة"
                      >
                        <CloseIcon />
                      </button>
                    </div>
                  </li>
                )
              })}
            </ul>
          )}
        </div>

        {/* قسم الأذكار المتاحة للإضافة */}
        <div className={styles.section}>
          <div className={styles.sectionHeader}>
            <h4 className={styles.sectionTitle}>
              أذكار يمكنك إضافتها <span className={styles.badge}>{filteredAvailableAdhkar.length}</span>
            </h4>
          </div>

          {/* شريط البحث في الأذكار المتاحة */}
          {allAdhkar.length > 5 && (
            <input
              type="text"
              className={styles.searchInput}
              placeholder="ابحث عن ذكر لإضافته..."
              value={searchAvailable}
              onChange={(e) => setSearchAvailable(e.target.value)}
            />
          )}

          {filteredAvailableAdhkar.length === 0 ? (
            <p className={styles.emptyHint}>
              {searchAvailable ? 'لا توجد نتائج مطابقة' : 'جميع أذكارك مضافة بالفعل في هذه المجموعة.'}
            </p>
          ) : (
            <ul className={`${styles.itemsList} ${styles.scrollable}`}>
              {filteredAvailableAdhkar.map((zekr) => (
                <li key={zekr.id} className={styles.availableItem}>
                  <span className={styles.itemText}>{zekr.text}</span>
                  <button
                    type="button"
                    className={styles.addBtn}
                    onClick={() => handleAdd(zekr.id)}
                    disabled={busyId === `add-${zekr.id}`}
                    title="إضافة للمجموعة"
                  >
                    <PlusIcon />
                    <span>إضافة</span>
                  </button>
                </li>
              ))}
            </ul>
          )}
        </div>
      </div>
    </Modal>
  )
}

export default GroupEditorModal

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'

// اقتراحات شائعة للمجموعات لتسهيل الإدخال السريع
const SUGGESTED_GROUPS = [
  'أذكار الصباح',
  'أذكار المساء',
  'أذكار النوم',
  'أذكار بعد الصلاة',
  'الورد اليومي',
]

function GroupFormModal({ isOpen, onClose, onConfirm, initial = null }) {
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [error, setError] = useState('')
  const [isSubmitting, setIsSubmitting] = useState(false)

  useEffect(() => {
    if (isOpen) {
      setName(initial?.name ?? '')
      setDescription(initial?.description ?? '')
      setError('')
      setIsSubmitting(false)
    }
  }, [isOpen, initial])

  const handleSubmit = async (e) => {
    e.preventDefault()
    const trimmedName = name.trim()

    if (!trimmedName) {
      setError('الرجاء إدخال اسم المجموعة')
      return
    }

    try {
      setIsSubmitting(true)
      await onConfirm(trimmedName, description.trim() || null)
    } catch (err) {
      setError(err?.message || 'فشل حفظ المجموعة، يرجى المحاولة ثانية')
    } finally {
      setIsSubmitting(false)
    }
  }

  const handleSelectSuggestion = (suggestedName) => {
    setName(suggestedName)
    if (error) setError('')
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={initial ? 'تعديل المجموعة' : 'إضافة مجموعة جديدة'}
    >
      <form className={styles.form} onSubmit={handleSubmit} noValidate>
        {/* حقل اسم المجموعة */}
        <div className={styles.fieldGroup}>
          <label htmlFor="group-name" className={styles.label}>
            <span>اسم المجموعة</span>
            <span className={styles.required}>*</span>
          </label>

          {/* اقتراحات سريعة (تظهر فقط عند إضافة مجموعة جديدة) */}
          {!initial && (
            <div className={styles.suggestions}>
              {SUGGESTED_GROUPS.map((item) => (
                <button
                  key={item}
                  type="button"
                  className={`${styles.suggestionChip} ${name === item ? styles.activeChip : ''}`}
                  onClick={() => handleSelectSuggestion(item)}
                >
                  {item}
                </button>
              ))}
            </div>
          )}

          <input
            id="group-name"
            className={styles.input}
            type="text"
            value={name}
            onChange={(e) => {
              setName(e.target.value)
              if (error) setError('')
            }}
            placeholder="مثال: أذكار الصباح..."
            maxLength={50}
            autoFocus
          />
        </div>

        {/* حقل وصف المجموعة */}
        <div className={styles.fieldGroup}>
          <div className={styles.labelRow}>
            <label htmlFor="group-desc" className={styles.label}>
              الوصف
            </label>
            <span className={styles.optionalHint}>اختياري</span>
          </div>

          <textarea
            id="group-desc"
            className={styles.textarea}
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            placeholder="وصف مختصر لمحتوى المجموعة أو وقت قراءتها..."
            rows={2}
            maxLength={150}
          />
        </div>

        {/* رسالة الخطأ */}
        {error && (
          <div className={styles.error} role="alert">
            <span>⚠️</span>
            <span>{error}</span>
          </div>
        )}

        {/* أزرار الإلغاء والحفظ */}
        <div className={styles.buttons}>
          <button
            type="button"
            className={styles.cancel}
            onClick={onClose}
            disabled={isSubmitting}
          >
            إلغاء
          </button>
          <button
            type="submit"
            className={styles.confirm}
            disabled={isSubmitting}
          >
            {isSubmitting ? 'جاري الحفظ...' : initial ? 'حفظ التعديلات' : 'إضافة'}
          </button>
        </div>
      </form>
    </Modal>
  )
}

function GroupsSection({
  groups = [],
  onAdd,
  onUpdate,
  onDelete,
  onAddItem,
  onUpdateItem,
  onDeleteItem,
  allAdhkar = [],
}) {
  const [formMode, setFormMode] = useState(null) // null | 'add' | { editing: group }
  const [deleteTarget, setDeleteTarget] = useState(null)
  const [editorGroupId, setEditorGroupId] = useState(null)
  const [isDeleting, setIsDeleting] = useState(false)

  const editorGroup = groups.find((g) => g.id === editorGroupId)

  const handleAdd = async (name, description) => {
    await onAdd(name, description)
    setFormMode(null)
  }

  const handleUpdate = async (name, description) => {
    await onUpdate(formMode.editing.id, name, description)
    setFormMode(null)
  }

  const handleDelete = async () => {
    if (!deleteTarget) return
    try {
      setIsDeleting(true)
      await onDelete(deleteTarget.id)
      setDeleteTarget(null)
    } catch (err) {
      console.error('فشل حذف المجموعة:', err)
    } finally {
      setIsDeleting(false)
    }
  }

  return (
    <SettingsSection title={`المجموعات (${groups.length})`}>
      <div className={styles.container}>
        {/* شريط الإجراءات العلوي */}
        <div className={styles.topBar}>
          <p className={styles.hint}>
            رتّب أذكارك في مجموعات (مثل: أذكار الصباح، بعد الصلاة) لتكرارها تلقائياً.
          </p>
          <button
            type="button"
            className={styles.addBtn}
            onClick={() => setFormMode('add')}
          >
            <span>+</span>
            <span>مجموعة جديدة</span>
          </button>
        </div>

        {/* عرض المجموعات */}
        {groups.length === 0 ? (
          <div className={styles.emptyCard}>
            <div className={styles.emptyIcon}>📂</div>
            <p className={styles.emptyTitle}>لا توجد مجموعات حتى الآن</p>
            <p className={styles.emptySub}>
              أنشئ مجموعتك الأولى واجمع أذكارك المفضلة للورد اليومي.
            </p>
          </div>
        ) : (
          <ul className={styles.list}>
            {groups.map((g) => {
              const count = g.items?.length ?? 0
              return (
                <li key={g.id} className={styles.item}>
                  {/* زر الدخول لمحرر أذكار المجموعة */}
                  <button
                    type="button"
                    className={styles.mainClickable}
                    onClick={() => setEditorGroupId(g.id)}
                    title="اضغط لترتيب وتعديل أذكار المجموعة"
                  >
                    <div className={styles.folderBadge}>
                      <FolderIcon />
                    </div>

                    <div className={styles.info}>
                      <div className={styles.titleLine}>
                        <span className={styles.name}>{g.name}</span>
                        <span className={styles.countBadge}>
                          {count === 0 ? 'فارغة' : `${count} ${count > 10 ? 'ذكر' : 'أذكار'}`}
                        </span>
                      </div>

                      {g.description && (
                        <p className={styles.description}>{g.description}</p>
                      )}
                    </div>

                    <span className={styles.openArrow}>
                      <ArrowLeftIcon />
                    </span>
                  </button>

                  {/* أزرار الإجراءات (تعديل الاسم / حذف المجموعة) */}
                  <div className={styles.actions}>
                    <button
                      type="button"
                      className={`${styles.actionBtn} ${styles.edit}`}
                      onClick={() => setFormMode({ editing: g })}
                      title="تعديل الاسم والوصف"
                      aria-label={`تعديل مجموعة ${g.name}`}
                    >
                      <EditIcon />
                    </button>
                    <button
                      type="button"
                      className={`${styles.actionBtn} ${styles.danger}`}
                      onClick={() => setDeleteTarget(g)}
                      title="حذف المجموعة"
                      aria-label={`حذف مجموعة ${g.name}`}
                    >
                      <DeleteIcon />
                    </button>
                  </div>
                </li>
              )
            })}
          </ul>
        )}
      </div>

      {/* نافذة إضافة / تعديل الاسم والوصف */}
      <GroupFormModal
        isOpen={formMode === 'add' || formMode?.editing != null}
        onClose={() => setFormMode(null)}
        onConfirm={formMode === 'add' ? handleAdd : handleUpdate}
        initial={formMode?.editing ?? null}
      />

      {/* نافذة تأكيد الحذف */}
      <ConfirmModal
        isOpen={deleteTarget !== null}
        onClose={() => setDeleteTarget(null)}
        onConfirm={handleDelete}
        title="حذف المجموعة؟"
        message={`سيتم حذف مجموعة "${deleteTarget?.name}" وكل محتوياتها نهائياً.`}
        confirmLabel={isDeleting ? 'جاري الحذف...' : 'نعم، احذف'}
        cancelLabel="إلغاء"
        danger
      />

      {/* نافذة تحرير محتوى وترتيب الأذكار */}
      <GroupEditorModal
        isOpen={editorGroupId !== null}
        onClose={() => setEditorGroupId(null)}
        group={editorGroup}
        allAdhkar={allAdhkar}
        onAddItem={onAddItem}
        onUpdateItem={onUpdateItem}
        onDeleteItem={onDeleteItem}
      />
    </SettingsSection>
  )
}

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './AdhkarFormModal.module.scss'

function AdhkarFormModal({ isOpen, onClose, onConfirm, initial = null }) {
  const [text, setText] = useState('')
  const [target, setTarget] = useState('')
  const [error, setError] = useState('')

  useEffect(() => {
    if (isOpen) {
      setText(initial?.text ?? '')
      setTarget(initial?.default_target ?? '')
      setError('')
    }
  }, [isOpen, initial])

  const handleConfirm = () => {
    const trimmedText = text.trim()
    if (!trimmedText) {
      setError('الرجاء إدخال نص الذكر')
      return
    }

    let targetValue = null
    if (target !== '') {
      const num = Number(target)
      if (isNaN(num) || num < 1 || num > 10000) {
        setError('الهدف يجب أن يكون بين 1 و 10000')
        return
      }
      targetValue = num
    }

    onConfirm(trimmedText, targetValue)
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={initial ? 'تعديل الذكر' : 'إضافة ذكر'}
    >
      <label className={styles.label}>
        نص الذكر
        <input
          className={styles.input}
          type="text"
          value={text}
          onChange={(e) => {
            setText(e.target.value)
            setError('')
          }}
          autoFocus
        />
      </label>

      <label className={styles.label}>
        الهدف (اختياري)
        <input
          className={styles.input}
          type="number"
          value={target}
          onChange={(e) => {
            setTarget(e.target.value)
            setError('')
          }}
          placeholder="اتركه فارغًا لذكر بدون هدف"
          min={1}
          max={10000}
        />
      </label>

      {error && <div className={styles.error}>{error}</div>}

      <div className={styles.buttons}>
        <button className={styles.cancel} onClick={onClose}>
          إلغاء
        </button>
        <button className={styles.confirm} onClick={handleConfirm}>
          {initial ? 'حفظ' : 'إضافة'}
        </button>
      </div>
    </Modal>
  )
}

import { useState } from 'react'
import SettingsSection from './SettingsSection.jsx'
import AdhkarFormModal from './AdhkarFormModal.jsx'
import ConfirmModal from '@/components/ui/ConfirmModal.jsx'
import styles from './AdhkarSection.module.scss'

function AdhkarSection({
  adhkar,
  onAdd,
  onUpdate,
  onDelete,
}) {
  const [formMode, setFormMode] = useState(null) // null | 'add' | { editing: zekr }
  const [deleteTarget, setDeleteTarget] = useState(null)

  const handleAdd = async (text, target) => {
    await onAdd(text, target)
    setFormMode(null)
  }

  const handleUpdate = async (text, target) => {
    await onUpdate(formMode.editing.id, text, target)
    setFormMode(null)
  }

  const handleDelete = async () => {
    await onDelete(deleteTarget.id)
    setDeleteTarget(null)
  }

  return (
    <SettingsSection title="الأذكار">
      <ul className={styles.list}>
        {adhkar.map((z) => (
          <li key={z.id} className={styles.item}>
            <span className={styles.text}>{z.text}</span>
            <div className={styles.actions}>
              <button
                className={styles.action}
                onClick={() => setFormMode({ editing: z })}
                aria-label="تعديل"
              >
                ✏️
              </button>
              <button
                className={`${styles.action} ${styles.danger}`}
                onClick={() => setDeleteTarget(z)}
                aria-label="حذف"
              >
                🗑️
              </button>
            </div>
          </li>
        ))}
      </ul>

      <button
        className={styles.addBtn}
        onClick={() => setFormMode('add')}
      >
        + إضافة ذكر
      </button>

      <AdhkarFormModal
        isOpen={formMode === 'add' || (formMode?.editing != null)}
        onClose={() => setFormMode(null)}
        onConfirm={formMode === 'add' ? handleAdd : handleUpdate}
        initial={formMode?.editing ?? null}
      />

      <ConfirmModal
        isOpen={deleteTarget !== null}
        onClose={() => setDeleteTarget(null)}
        onConfirm={handleDelete}
        title="حذف الذكر؟"
        message={`سيتم حذف "${deleteTarget?.text}" مع كل بياناته (الأعداد والأهداف والإحصائيات). لا يمكن التراجع.`}
        confirmLabel="نعم، احذف"
        cancelLabel="إلغاء"
        danger
      />
    </SettingsSection>
  )
}

function DataSection({ onDeleteCounts, onDeleteTargets, onDeleteClicks, onDeleteAll }) {
  const [confirm, setConfirm] = useState(null)  // null | 'counts' | 'targets' | 'clicks' | 'all'
  const [loading, setLoading] = useState(false)

  const openConfirm = (type) => setConfirm(type)
  const closeConfirm = () => setConfirm(null)

  const handleConfirm = async () => {
    setLoading(true)
    try {
      if (confirm === 'counts') await onDeleteCounts()
      else if (confirm === 'targets') await onDeleteTargets()
      else if (confirm === 'clicks') await onDeleteClicks()
      else if (confirm === 'all') await onDeleteAll()

      closeConfirm()
    } catch (err) {
      console.error('فشل المسح:', err)
    } finally {
      setLoading(false)
    }
  }

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './GroupEditorModal.module.scss'

function GroupEditorModal({
  isOpen,
  onClose,
  group,
  allAdhkar = [],
  onAddItem,
  onUpdateItem,
  onDeleteItem,
}) {
  // state محلي لتعديل الهدف
  const [editingTargetId, setEditingTargetId] = useState(null)
  const [tempTarget, setTempTarget] = useState('')

  // إعادة تعيين عند الإغلاق
  useEffect(() => {
    if (!isOpen) {
      setEditingTargetId(null)
      setTempTarget('')
    }
  }, [isOpen])

  if (!group) return null

  // الأذكار المتاحة (مش في المجموعة)
  const usedIds = new Set(group.items.map((i) => i.zekr_id))
  const availableAdhkar = allAdhkar.filter((z) => !usedIds.has(z.id))

  // ===========================
  // Handlers
  // ===========================

  const handleAdd = async (zekr_id) => {
    const zekr = allAdhkar.find((z) => z.id === zekr_id)
    const target = zekr?.default_target ?? null
    await onAddItem(group.id, zekr_id, target)
  }

  const handleDelete = async (itemId) => {
    await onDeleteItem(group.id, itemId)
  }

  // ترتيب: فوق
  const handleMoveUp = async (item, index) => {
    if (index === 0) return
    const prev = group.items[index - 1]

    // نبدّل الـ positions
    await onUpdateItem(group.id, item.id, { position: prev.position })
    await onUpdateItem(group.id, prev.id, { position: item.position })
  }

  // ترتيب: تحت
  const handleMoveDown = async (item, index) => {
    if (index === group.items.length - 1) return
    const next = group.items[index + 1]

    await onUpdateItem(group.id, item.id, { position: next.position })
    await onUpdateItem(group.id, next.id, { position: item.position })
  }

  // تعديل الهدف
  const handleEditTarget = (item) => {
    setEditingTargetId(item.id)
    setTempTarget(String(item.target ?? item.default_target ?? ''))
  }

  const handleSaveTarget = async (item) => {
    const num = Number(tempTarget)
    if (isNaN(num) || num < 1 || num > 10000) {
      alert('الرجاء إدخال رقم صحيح بين 1 و 10000')
      return
    }
    await onUpdateItem(group.id, item.id, { target: num })
    setEditingTargetId(null)
    setTempTarget('')
  }

  const handleCancelTarget = () => {
    setEditingTargetId(null)
    setTempTarget('')
  }

  const handleTargetKeyDown = (e, item) => {
    if (e.key === 'Enter') handleSaveTarget(item)
    if (e.key === 'Escape') handleCancelTarget()
  }

  return (
    <Modal isOpen={isOpen} onClose={onClose} title={group.name}>
      <div className={styles.body}>
        {group.description && (
          <p className={styles.description}>{group.description}</p>
        )}

        {/* ===========================
            أذكار المجموعة
        =========================== */}
        <div className={styles.section}>
          <h4 className={styles.sectionTitle}>
            الأذكار ({group.items.length})
          </h4>

          {group.items.length === 0 ? (
            <p className={styles.empty}>لا توجد أذكار بعد</p>
          ) : (
            <ul className={styles.itemsList}>
              {group.items.map((item, index) => (
                <li key={item.id} className={styles.item}>
                  <div className={styles.itemPosition}>{index + 1}</div>

                  <div className={styles.itemInfo}>
                    <div className={styles.itemText}>{item.text}</div>

                    {editingTargetId === item.id ? (
                      <div className={styles.targetEdit}>
                        <input
                          className={styles.targetInput}
                          type="number"
                          value={tempTarget}
                          onChange={(e) => setTempTarget(e.target.value)}
                          onKeyDown={(e) => handleTargetKeyDown(e, item)}
                          autoFocus
                          min={1}
                          max={10000}
                        />
                        <button
                          className={styles.saveBtn}
                          onClick={() => handleSaveTarget(item)}
                          aria-label="حفظ"
                        >
                          ✓
                        </button>
                        <button
                          className={styles.cancelBtn}
                          onClick={handleCancelTarget}
                          aria-label="إلغاء"
                        >
                          ✕
                        </button>
                      </div>
                    ) : (
                      <button
                        className={styles.targetDisplay}
                        onClick={() => handleEditTarget(item)}
                      >
                        الهدف: {item.target ?? item.default_target ?? '—'}
                      </button>
                    )}
                  </div>

                  <div className={styles.itemActions}>
                    <button
                      className={styles.moveBtn}
                      onClick={() => handleMoveUp(item, index)}
                      disabled={index === 0}
                      aria-label="فوق"
                    >
                      ↑
                    </button>
                    <button
                      className={styles.moveBtn}
                      onClick={() => handleMoveDown(item, index)}
                      disabled={index === group.items.length - 1}
                      aria-label="تحت"
                    >
                      ↓
                    </button>
                    <button
                      className={styles.deleteBtn}
                      onClick={() => handleDelete(item.id)}
                      aria-label="حذف"
                    >
                      ✕
                    </button>
                  </div>
                </li>
              ))}
            </ul>
          )}
        </div>

        {/* ===========================
            الأذكار المتاحة
        =========================== */}
        <div className={styles.section}>
          <h4 className={styles.sectionTitle}>
            الأذكار المتاحة ({availableAdhkar.length})
          </h4>

          {availableAdhkar.length === 0 ? (
            <p className={styles.empty}>كل الأذكار مضافة بالفعل</p>
          ) : (
            <ul className={styles.itemsList}>
              {availableAdhkar.map((zekr) => (
                <li key={zekr.id} className={styles.item}>
                  <div className={styles.itemInfo}>
                    <div className={styles.itemText}>{zekr.text}</div>
                  </div>
                  <button
                    className={styles.addBtn}
                    onClick={() => handleAdd(zekr.id)}
                    aria-label="إضافة"
                  >
                    +
                  </button>
                </li>
              ))}
            </ul>
          )}
        </div>
      </div>
    </Modal>
  )
}

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './GroupFormModal.module.scss'

function GroupFormModal({ isOpen, onClose, onConfirm, initial = null }) {
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [error, setError] = useState('')

  useEffect(() => {
    if (isOpen) {
      setName(initial?.name ?? '')
      setDescription(initial?.description ?? '')
      setError('')
    }
  }, [isOpen, initial])

  const handleConfirm = () => {
    const trimmed = name.trim()
    if (!trimmed) {
      setError('الرجاء إدخال اسم المجموعة')
      return
    }
    onConfirm(trimmed, description.trim() || null)
  }

  const handleKeyDown = (e) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault()
      handleConfirm()
    }
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={initial ? 'تعديل المجموعة' : 'إضافة مجموعة'}
    >
      <label className={styles.label}>
        الاسم
        <input
          className={styles.input}
          type="text"
          value={name}
          onChange={(e) => {
            setName(e.target.value)
            setError('')
          }}
          onKeyDown={handleKeyDown}
          autoFocus
        />
      </label>

      <label className={styles.label}>
        الوصف (اختياري)
        <textarea
          className={styles.textarea}
          value={description}
          onChange={(e) => setDescription(e.target.value)}
          rows={2}
        />
      </label>

      {error && <div className={styles.error}>{error}</div>}

      <div className={styles.buttons}>
        <button className={styles.cancel} onClick={onClose}>
          إلغاء
        </button>
        <button className={styles.confirm} onClick={handleConfirm}>
          {initial ? 'حفظ' : 'إضافة'}
        </button>
      </div>
    </Modal>
  )
}

import ConfirmModal from '@/components/ui/ConfirmModal.jsx'

function GroupsSection({
  groups,
  onAdd,
  onUpdate,
  onDelete,
  onAddItem,
  onUpdateItem,
  onDeleteItem,
  allAdhkar,
}) {
  const [formMode, setFormMode] = useState(null) // null | 'add' | { editing: group }
  const [deleteTarget, setDeleteTarget] = useState(null)
  const [editorGroupId, setEditorGroupId] = useState(null)

  const editorGroup = groups.find((g) => g.id === editorGroupId)

  const handleAdd = async (name, description) => {
    await onAdd(name, description)
    setFormMode(null)
  }

  const handleUpdate = async (name, description) => {
    await onUpdate(formMode.editing.id, name, description)
    setFormMode(null)
  }

  const handleDelete = async () => {
    await onDelete(deleteTarget.id)
    setDeleteTarget(null)
  }

  return (
    <SettingsSection title="المجموعات">
      {groups.length === 0 ? (
        <p className={styles.empty}>لا توجد مجموعات بعد</p>
      ) : (
        <ul className={styles.list}>
          {groups.map((g) => (
            <li key={g.id} className={styles.item}>
              <button
                className={styles.groupBtn}
                onClick={() => setEditorGroupId(g.id)}
              >
                <span className={styles.groupName}>{g.name}</span>
                <span className={styles.groupCount}>
                  {g.items?.length ?? 0} ذكر
                </span>
              </button>

              <div className={styles.actions}>
                <button
                  className={styles.action}
                  onClick={() => setFormMode({ editing: g })}
                  aria-label="تعديل"
                >
                  ✏️
                </button>
                <button
                  className={`${styles.action} ${styles.danger}`}
                  onClick={() => setDeleteTarget(g)}
                  aria-label="حذف"
                >
                  🗑️
                </button>
              </div>
            </li>
          ))}
        </ul>
      )}

      <button className={styles.addBtn} onClick={() => setFormMode('add')}>
        + إضافة مجموعة
      </button>

      <GroupFormModal
        isOpen={formMode === 'add' || formMode?.editing != null}
        onClose={() => setFormMode(null)}
        onConfirm={formMode === 'add' ? handleAdd : handleUpdate}
        initial={formMode?.editing ?? null}
      />

      <ConfirmModal
        isOpen={deleteTarget !== null}
        onClose={() => setDeleteTarget(null)}
        onConfirm={handleDelete}
        title="حذف المجموعة؟"
        message={`سيتم حذف "${deleteTarget?.name}" وكل أذكارها. لا يمكن التراجع.`}
        confirmLabel="نعم، احذف"
        cancelLabel="إلغاء"
        danger
      />

      <GroupEditorModal
        isOpen={editorGroupId !== null}
        onClose={() => setEditorGroupId(null)}
        group={editorGroup}
        allAdhkar={allAdhkar}
        onAddItem={onAddItem}
        onUpdateItem={onUpdateItem}
        onDeleteItem={onDeleteItem}
      />
    </SettingsSection>
  )
}
