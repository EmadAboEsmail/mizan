function GroupsSection() {
  const {
    groups,
    allAdhkar,
    handleAddGroup,
    handleUpdateGroup,
    handleDeleteGroup,
    handleAddGroupItem,
    handleUpdateGroupItem,
    handleDeleteGroupItem,
  } = useZekrData()

  const [formMode, setFormMode] = useState(null) // null | 'add' | { editing: group }
  const [deleteTarget, setDeleteTarget] = useState(null)
  const [editorGroupId, setEditorGroupId] = useState(null)
  const [isDeleting, setIsDeleting] = useState(false)

  // ✅ بحث آمن بمقارنة String() — يدعم id نص أو رقم
  const editorGroup = useMemo(() => {
    if (editorGroupId == null) return null
    return (
      groups.find((g) => String(g.id) === String(editorGroupId)) ?? null
    )
  }, [editorGroupId, groups])

  // ✅ فتح المحرر (دالة موحدة)
  const openEditor = (group) => {
    if (!group) return
    console.log('[GroupsSection] Opening editor for:', group.id, group.name)
    setEditorGroupId(group.id)
  }

  const handleAdd = async (name, description) => {
    await handleAddGroup(name, description)
    setFormMode(null)
  }

  const handleUpdate = async (name, description) => {
    await handleUpdateGroup(formMode.editing.id, name, description)
    setFormMode(null)
  }

  const handleDelete = async () => {
    if (!deleteTarget) return
    try {
      setIsDeleting(true)
      await handleDeleteGroup(deleteTarget.id)
      setDeleteTarget(null)
    } catch (err) {
      console.error('فشل حذف المجموعة:', err)
    } finally {
      setIsDeleting(false)
    }
  }

  return (
    <SettingsSection id="groups" title={`المجموعات (${groups.length})`}>
      <div className={styles.container}>
        {/* شريط الإجراءات العلوي */}
        <div className={styles.topBar}>
          <p className={styles.hint}>
            رتّب أذكارك في مجموعات (مثل: أذكار الصباح، بعد الصلاة) لتكرارها
            تلقائياً.
          </p>
          <button
            type="button"
            className={styles.addBtn}
            onClick={() => setFormMode('add')}
          >
            <span>+</span>
            <span>مجموعة جديدة</span>
          </button>
        </div>

        {/* عرض المجموعات */}
        {groups.length === 0 ? (
          <div className={styles.emptyCard}>
            <div className={styles.emptyIcon}>📂</div>
            <p className={styles.emptyTitle}>لا توجد مجموعات حتى الآن</p>
            <p className={styles.emptySub}>
              أنشئ مجموعتك الأولى واجمع أذكارك المفضلة للورد اليومي.
            </p>
          </div>
        ) : (
          <ul className={styles.list}>
            {groups.map((g) => {
              const count = g.items?.length ?? 0
              return (
                <li
                  key={g.id}
                  className={styles.item}
                  onClick={() => openEditor(g)}
                  onKeyDown={(e) => {
                    if (e.key === 'Enter' || e.key === ' ') {
                      e.preventDefault()
                      openEditor(g)
                    }
                  }}
                  role="button"
                  tabIndex={0}
                  aria-label={`فتح مجموعة ${g.name} لترتيب الأذكار`}
                >
                  <div className={styles.mainClickable}>
                    <div className={styles.folderBadge}>
                      <FolderIcon />
                    </div>

                    <div className={styles.info}>
                      <div className={styles.titleLine}>
                        <span className={styles.name}>{g.name}</span>
                        <span className={styles.countBadge}>
                          {count === 0
                            ? 'فارغة'
                            : `${count} ${count > 10 ? 'ذكر' : 'أذكار'}`}
                        </span>
                      </div>

                      {g.description && (
                        <p className={styles.description}>{g.description}</p>
                      )}
                    </div>

                    {/* ✅ السهم أصبح قابل للنقر مباشرة (fallback) */}
                    <span
                      className={styles.openArrow}
                      onClick={(e) => {
                        e.stopPropagation()
                        openEditor(g)
                      }}
                      role="button"
                      tabIndex={-1}
                      aria-hidden="true"
                    >
                      <ArrowLeftIcon />
                    </span>
                  </div>

                  {/* ✅ أزرار الإجراءات مع stopPropagation */}
                  <div
                    className={styles.actions}
                    onClick={(e) => e.stopPropagation()}
                    onKeyDown={(e) => e.stopPropagation()}
                  >
                    <button
                      type="button"
                      className={`${styles.actionBtn} ${styles.edit}`}
                      onClick={(e) => {
                        e.stopPropagation()
                        setFormMode({ editing: g })
                      }}
                      title="تعديل الاسم والوصف"
                      aria-label={`تعديل مجموعة ${g.name}`}
                    >
                      <EditIcon />
                    </button>
                    <button
                      type="button"
                      className={`${styles.actionBtn} ${styles.danger}`}
                      onClick={(e) => {
                        e.stopPropagation()
                        setDeleteTarget(g)
                      }}
                      title="حذف المجموعة"
                      aria-label={`حذف مجموعة ${g.name}`}
                    >
                      <DeleteIcon />
                    </button>
                  </div>
                </li>
              )
            })}
          </ul>
        )}
      </div>

      {/* نافذة إضافة / تعديل الاسم والوصف */}
      <GroupFormModal
        isOpen={formMode === 'add' || formMode?.editing != null}
        onClose={() => setFormMode(null)}
        onConfirm={formMode === 'add' ? handleAdd : handleUpdate}
        initial={formMode?.editing ?? null}
      />

      {/* نافذة تأكيد الحذف */}
      <ConfirmModal
        isOpen={deleteTarget !== null}
        onClose={() => setDeleteTarget(null)}
        onConfirm={handleDelete}
        title="حذف المجموعة؟"
        message={`سيتم حذف مجموعة "${deleteTarget?.name}" وكل محتوياتها نهائياً.`}
        confirmLabel={isDeleting ? 'جاري الحذف...' : 'نعم، احذف'}
        cancelLabel="إلغاء"
        danger
      />

      {/* نافذة تحرير محتوى وترتيب الأذكار */}
      <GroupEditorModal
        isOpen={editorGroupId !== null}
        onClose={() => setEditorGroupId(null)}
        group={editorGroup}
        allAdhkar={allAdhkar}
        onAddItem={handleAddGroupItem}
        onUpdateItem={handleUpdateGroupItem}
        onDeleteItem={handleDeleteGroupItem}
      />
    </SettingsSection>
  )
}

export default GroupsSection

import { useState, useEffect } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './GroupFormModal.module.scss'

// اقتراحات شائعة للمجموعات لتسهيل الإدخال السريع
const SUGGESTED_GROUPS = [
  'أذكار الصباح',
  'أذكار المساء',
  'أذكار النوم',
  'أذكار بعد الصلاة',
  'الورد اليومي',
]

function GroupFormModal({ isOpen, onClose, onConfirm, initial = null }) {
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [error, setError] = useState('')
  const [isSubmitting, setIsSubmitting] = useState(false)

  useEffect(() => {
    if (isOpen) {
      setName(initial?.name ?? '')
      setDescription(initial?.description ?? '')
      setError('')
      setIsSubmitting(false)
    }
  }, [isOpen, initial])

  const handleSubmit = async (e) => {
    e.preventDefault()
    const trimmedName = name.trim()

    if (!trimmedName) {
      setError('الرجاء إدخال اسم المجموعة')
      return
    }

    try {
      setIsSubmitting(true)
      await onConfirm(trimmedName, description.trim() || null)
    } catch (err) {
      setError(err?.message || 'فشل حفظ المجموعة، يرجى المحاولة ثانية')
    } finally {
      setIsSubmitting(false)
    }
  }

  const handleSelectSuggestion = (suggestedName) => {
    setName(suggestedName)
    if (error) setError('')
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={initial ? 'تعديل المجموعة' : 'إضافة مجموعة جديدة'}
    >
      <form className={styles.form} onSubmit={handleSubmit} noValidate>
        {/* حقل اسم المجموعة */}
        <div className={styles.fieldGroup}>
          <label htmlFor="group-name" className={styles.label}>
            <span>اسم المجموعة</span>
            <span className={styles.required}>*</span>
          </label>

          {/* اقتراحات سريعة (تظهر فقط عند إضافة مجموعة جديدة) */}
          {!initial && (
            <div className={styles.suggestions}>
              {SUGGESTED_GROUPS.map((item) => (
                <button
                  key={item}
                  type="button"
                  className={`${styles.suggestionChip} ${name === item ? styles.activeChip : ''
                    }`}
                  onClick={() => handleSelectSuggestion(item)}
                >
                  {item}
                </button>
              ))}
            </div>
          )}

          <input
            id="group-name"
            className={styles.input}
            type="text"
            value={name}
            onChange={(e) => {
              setName(e.target.value)
              if (error) setError('')
            }}
            placeholder="مثال: أذكار الصباح..."
            maxLength={50}
            autoFocus
          />
        </div>

        {/* حقل وصف المجموعة */}
        <div className={styles.fieldGroup}>
          <div className={styles.labelRow}>
            <label htmlFor="group-desc" className={styles.label}>
              الوصف
            </label>
            <span className={styles.optionalHint}>اختياري</span>
          </div>

          <textarea
            id="group-desc"
            className={styles.textarea}
            value={description}
            onChange={(e) => setDescription(e.target.value)}
            placeholder="وصف مختصر لمحتوى المجموعة أو وقت قراءتها..."
            rows={2}
            maxLength={150}
          />
        </div>

        {/* رسالة الخطأ */}
        {error && (
          <div className={styles.error} role="alert">
            <span>⚠️</span>
            <span>{error}</span>
          </div>
        )}

        {/* أزرار الإلغاء والحفظ */}
        <div className={styles.buttons}>
          <button
            type="button"
            className={styles.cancel}
            onClick={onClose}
            disabled={isSubmitting}
          >
            إلغاء
          </button>
          <button
            type="submit"
            className={styles.confirm}
            disabled={isSubmitting}
          >
            {isSubmitting
              ? 'جاري الحفظ...'
              : initial
                ? 'حفظ التعديلات'
                : 'إضافة'}
          </button>
        </div>
      </form>
    </Modal>
  )
}

export default GroupFormModal

import { useState, useEffect, useMemo } from 'react'
import Modal from '@/components/ui/Modal.jsx'
import styles from './GroupEditorModal.module.scss'

// أيقونات SVG خفيفة
const ArrowUpIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="18 15 12 9 6 15" />
  </svg>
)

const ArrowDownIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="6 9 12 15 18 9" />

const CheckIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <polyline points="20 6 9 17 4 12" />
  </svg>
)

const CloseIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <line x1="18" y1="6" x2="6" y2="18" />

  </svg>
)

const PlusIcon = () => (
  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
    <line x1="12" y1="5" x2="12" y2="19" />
    <line x1="5" y1="12" x2="19" y2="12" />
  </svg>
)

function GroupEditorModal({
  isOpen,
  onClose,
  group,
  allAdhkar = [],
  onAddItem,
  onUpdateItem,
  onDeleteItem,
}) {
  const [editingTargetId, setEditingTargetId] = useState(null)
  const [tempTarget, setTempTarget] = useState('')
  const [targetError, setTargetError] = useState('')
  const [searchAvailable, setSearchAvailable] = useState('')
  const [busyId, setBusyId] = useState(null)

  useEffect(() => {
    if (!isOpen) {
      setEditingTargetId(null)
      setTempTarget('')
      setTargetError('')
      setSearchAvailable('')
      setBusyId(null)
    }
  }, [isOpen])

  // حماية آمنة لعناصر المجموعة
  const items = group?.items ?? []

  // استخراج الأذكار المتاحة وفلترتها بالبحث
  const usedIds = useMemo(() => new Set(items.map((i) => i.zekr_id)), [items])

  const filteredAvailableAdhkar = useMemo(() => {
    const available = allAdhkar.filter((z) => !usedIds.has(z.id))
    if (!searchAvailable.trim()) return available
    return available.filter((z) =>
      z.text.toLowerCase().includes(searchAvailable.trim().toLowerCase())
    )
  }, [allAdhkar, usedIds, searchAvailable])

  // ✅ رسالة تشخيصية لو المجموعة مش موجودة بدل الاختفاء الصامت
  if (!group) {
    if (!isOpen) return null
    return (
      <Modal isOpen={isOpen} onClose={onClose} title={group.title} >
        <div
          style={{
            padding: '2rem 1rem',
            textAlign: 'center',
            color: 'var(--text-muted)',
          }}
        >
          <div style={{ fontSize: '2.5rem', marginBottom: '0.75rem' }}>
            ⚠️
          </div>
          <p style={{ margin: 0, fontWeight: 700 }}>
            تعذّر فتح المجموعة
          </p>
          <p style={{ margin: '0.5rem 0 0', fontSize: '0.85rem' }}>
            المجموعة غير موجودة أو تم حذفها. أعد تحميل الصفحة.
          </p>
        </div>
      </Modal>
    )
  }

  // ===========================
  // Handlers
  // ===========================

  const handleAdd = async (zekr_id) => {
    setBusyId(`add-${zekr_id}`)
    try {
      const zekr = allAdhkar.find((z) => z.id === zekr_id)
      const target = zekr?.default_target ?? null
      await onAddItem(group.id, zekr_id, target)
    } finally {
      setBusyId(null)
    }
  }

  const handleDelete = async (itemId) => {
    setBusyId(`del-${itemId}`)
    try {
      await onDeleteItem(group.id, itemId)
    } finally {
      setBusyId(null)
    }
  }

  const handleMoveUp = async (item, index) => {
    if (index === 0 || busyId) return
    const prev = items[index - 1]
    setBusyId(`move-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, {
        position: prev.position ?? index - 1,
      })
      await onUpdateItem(group.id, prev.id, {
        position: item.position ?? index,
      })
    } finally {
      setBusyId(null)
    }
  }

  const handleMoveDown = async (item, index) => {
    if (index === items.length - 1 || busyId) return
    const next = items[index + 1]
    setBusyId(`move-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, {
        position: next.position ?? index + 1,
      })
      await onUpdateItem(group.id, next.id, {
        position: item.position ?? index,
      })
    } finally {
      setBusyId(null)
    }
  }

  const handleEditTarget = (item) => {
    setEditingTargetId(item.id)
    setTempTarget(item.target != null ? String(item.target) : '')
    setTargetError('')
  }

  const handleSaveTarget = async (item) => {
    let finalTarget = null

    if (tempTarget.trim() !== '') {
      const num = Number(tempTarget)
      if (!Number.isInteger(num) || num < 1 || num > 10000) {
        setTargetError('بين 1 و 10,000')
        return
      }
      finalTarget = num
    }

    setBusyId(`target-${item.id}`)
    try {
      await onUpdateItem(group.id, item.id, { target: finalTarget })
      setEditingTargetId(null)
      setTempTarget('')
      setTargetError('')
    } finally {
      setBusyId(null)
    }
  }

  const handleCancelTarget = () => {
    setEditingTargetId(null)
    setTempTarget('')
    setTargetError('')
  }

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title={group.name}
      variant="sheet"
    >
      <div className={styles.body}>
        {group.description && (
          <p className={styles.description}>{group.description}</p>
        )}

        {/* قسم أذكار المجموعة الحالية */}
        <div className={styles.section}>
          <div className={styles.sectionHeader}>
            <h4 className={styles.sectionTitle}>
              أذكار المجموعة{' '}
              <span className={styles.badge}>{items.length}</span>
            </h4>
          </div>

          {items.length === 0 ? (
            <div className={styles.emptyCard}>
              لا توجد أذكار مضافة لهذه المجموعة حتى الآن.
            </div>
          ) : (
            <ul className={styles.itemsList}>
              {items.map((item, index) => {
                const isCurrentBusy =
                  busyId === `move-${item.id}` || busyId === `del-${item.id}`
                return (
                  <li
                    key={item.id}
                    className={`${styles.item} ${isCurrentBusy ? styles.isBusy : ''
                      }`}
                  >
                    {/* رقم الترتيب */}
                    <div className={styles.positionBadge}>{index + 1}</div>

                    <div className={styles.itemInfo}>
                      <span className={styles.itemText}>{item.text}</span>

                      {/* تعديل الهدف أو عرضه */}
                      {editingTargetId === item.id ? (
                        <div className={styles.targetEditGroup}>
                          <input
                            className={`${styles.targetInput} ${targetError ? styles.hasError : ''
                              }`}
                            type="number"
                            value={tempTarget}
                            onChange={(e) => {
                              setTempTarget(e.target.value)
                              if (targetError) setTargetError('')
                            }}
                            onKeyDown={(e) => {
                              if (e.key === 'Enter') handleSaveTarget(item)
                              if (e.key === 'Escape') handleCancelTarget()
                            }}
                            autoFocus
                            placeholder="افتراضي"
                            min={1}
                            max={10000}
                          />
                          <button
                            type="button"
                            className={styles.saveBtn}
                            onClick={() => handleSaveTarget(item)}
                            title="حفظ"
                          >
                            <CheckIcon />
                          </button>
                          <button
                            type="button"
                            className={styles.cancelBtn}
                            onClick={handleCancelTarget}
                            title="إلغاء"
                          >
                            <CloseIcon />
                          </button>
                          {targetError && (
                            <span className={styles.inlineError}>
                              {targetError}
                            </span>
                          )}
                        </div>
                      ) : (
                        <button
                          type="button"
                          className={styles.targetDisplay}
                          onClick={() => handleEditTarget(item)}
                          title="اضغط لتعديل هدف هذا الذكر"
                        >
                          🎯 الهدف:{' '}
                          {item.target ?? item.default_target ?? 'مفتوح'}
                        </button>
                      )}
                    </div>

                    {/* أزرار الترتيب والحذف */}
                    <div className={styles.itemActions}>
                      <button
                        type="button"
                        className={styles.iconBtn}
                        onClick={() => handleMoveUp(item, index)}
                        disabled={index === 0 || !!busyId}
                        title="تحريك لأعلى"
                      >
                        <ArrowUpIcon />
                      </button>
                      <button
                        type="button"
                        className={styles.iconBtn}
                        onClick={() => handleMoveDown(item, index)}
                        disabled={index === items.length - 1 || !!busyId}
                        title="تحريك لأسفل"
                      >
                        <ArrowDownIcon />
                      </button>
                      <button
                        type="button"
                        className={`${styles.iconBtn} ${styles.danger}`}
                        onClick={() => handleDelete(item.id)}
                        disabled={!!busyId}
                        title="إزالة من المجموعة"
                      >
                        <CloseIcon />
                      </button>
                    </div>
                  </li>
                )
              })}
            </ul>
          )}
        </div>

        {/* قسم الأذكار المتاحة للإضافة */}
        <div className={styles.section}>
          <div className={styles.sectionHeader}>
            <h4 className={styles.sectionTitle}>
              أذكار يمكنك إضافتها{' '}
              <span className={styles.badge}>
                {filteredAvailableAdhkar.length}
              </span>
            </h4>
          </div>

          {/* شريط البحث في الأذكار المتاحة */}
          {allAdhkar.length > 5 && (
            <input
              type="text"
              className={styles.searchInput}
              placeholder="ابحث عن ذكر لإضافته..."
              value={searchAvailable}
              onChange={(e) => setSearchAvailable(e.target.value)}
            />
          )}

          {filteredAvailableAdhkar.length === 0 ? (
            <p className={styles.emptyHint}>
              {searchAvailable
                ? 'لا توجد نتائج مطابقة'
                : 'جميع أذكارك مضافة بالفعل في هذه المجموعة.'}
            </p>
          ) : (
            <ul className={`${styles.itemsList} ${styles.scrollable}`}>
              {filteredAvailableAdhkar.map((zekr) => (
                <li key={zekr.id} className={styles.availableItem}>
                  <span className={styles.itemText}>{zekr.text}</span>
                  <button
                    type="button"
                    className={styles.addBtn}
                    onClick={() => handleAdd(zekr.id)}
                    disabled={busyId === `add-${zekr.id}`}
                    title="إضافة للمجموعة"
                  >
                    <PlusIcon />
                    <span>إضافة</span>
                  </button>
                </li>
              ))}
            </ul>
          )}
        </div>
      </div>
    </Modal>
  )
}

export default GroupEditorModal
